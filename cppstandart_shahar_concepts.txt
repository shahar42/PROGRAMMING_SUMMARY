// Function with a local static object 'A'
int& getCounter() {
    static int A = 0; // Local static
    return ++A;
}

// Another static object 'B'
struct UsesCounter {
    ~UsesCounter() {
        // This runs at program exit
        getCounter(); // Tries to use 'A'
    }
};

UsesCounter B; // Global static 'B'

int main() {
    // 1. 'B' is initialized (before main).
    getCounter(); // 2. 'A' is initialized.

    // At program exit:
    // 1. 'A' is destroyed (initialized last).
    // 2. 'B' is destroyed (initialized first).
    // 3. ~UsesCounter() calls getCounter().
    // 4. Control passes 'static int A = 0', which is already destroyed.
    // 5. This is UNDEFINED BEHAVIOR.
}
